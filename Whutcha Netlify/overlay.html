    <script>

        const supabaseUrl = 'https://kjoffilncbybquvjytxx.supabase.co';

        const supabaseKey = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Imtqb2ZmaWxuY2J5YnF1dmp5dHh4Iiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODEzNzM1MjgsImV4cCI6MjA5Njk0OTUyOH0.C7VSZspoYz_Qqc_cZBUCxA1citn301hub1o8NFcUefU';

        const supabase = window.supabase.createClient(supabaseUrl, supabaseKey);


        const widgetContainer = document.getElementById('widget-container');

        const mediaContainer = document.getElementById('media-container');

        const alertSound = document.getElementById('alert-sound');

        

        let currentRoom = null; 

        let currentScale = 1.0; // Храним текущий масштаб


        // 🔥 НОВАЯ ФУНКЦИЯ ДЛЯ УПРАВЛЕНИЯ ПОЗИЦИЕЙ С АЙФОНА

        function applyLayout(position, scale) {

            currentScale = scale || 1.0;

            widgetContainer.style.transform = `scale(${currentScale})`;

            

            // Сбрасываем координаты

            widgetContainer.style.top = 'auto';

            widgetContainer.style.bottom = 'auto';

            widgetContainer.style.left = 'auto';

            widgetContainer.style.right = 'auto';


            const pad = '40px';


            if (position === 'top-left') {

                widgetContainer.style.top = pad; widgetContainer.style.left = pad;

                widgetContainer.style.transformOrigin = 'top left';

            } else if (position === 'top-right') {

                widgetContainer.style.top = pad; widgetContainer.style.right = pad;

                widgetContainer.style.transformOrigin = 'top right';

            } else if (position === 'bottom-left') {

                widgetContainer.style.bottom = pad; widgetContainer.style.left = pad;

                widgetContainer.style.transformOrigin = 'bottom left';

            } else {

                widgetContainer.style.bottom = pad; widgetContainer.style.right = pad;

                widgetContainer.style.transformOrigin = 'bottom right';

            }

        }


        function showWidget(donation) {

            document.getElementById('donor-name').innerText = donation.username;

            document.getElementById('donor-initial').innerText = donation.username.charAt(0).toUpperCase();

            document.getElementById('donor-amount').innerText = `+$${donation.amount.toFixed(2)}`;

            

            const isCoStream = donation.video_hash === "costream_request";

            document.getElementById('donation-type').innerText = isCoStream ? "Live Co-Stream" : "Video Message";

            document.getElementById('donation-type').className = isCoStream 

                ? "text-xs text-purple-400 font-bold uppercase tracking-widest" 

                : "text-xs text-cyan-400 font-bold uppercase tracking-widest";


            widgetContainer.classList.remove('slide-out-anim');

            widgetContainer.classList.add('slide-in-anim');

            widgetContainer.style.display = 'flex';

            

            alertSound.volume = 0.5;

            alertSound.play().catch(e => console.log("Sound autoplay blocked by browser"));

        }


        function hideWidget() {

            widgetContainer.classList.remove('slide-in-anim');

            widgetContainer.classList.add('slide-out-anim');

            setTimeout(() => {

                widgetContainer.style.display = 'none';

                mediaContainer.innerHTML = ''; 

                if (currentRoom) {

                    currentRoom.disconnect();

                    currentRoom = null;

                }

            }, 500); 

        }


        function playVideoMessage(videoHash) {

            mediaContainer.innerHTML = ''; 

            const videoUrl = `${supabaseUrl}/storage/v1/object/public/videos/${videoHash}`;

            

            const videoEl = document.createElement('video');

            videoEl.src = videoUrl;

            videoEl.autoplay = true;

            videoEl.controls = false;

            

            videoEl.onended = () => {

                hideWidget(); 

            };

            

            mediaContainer.appendChild(videoEl);

        }


        async function connectCoStream(token) {

            mediaContainer.innerHTML = '<div class="text-white text-lg animate-pulse">Connecting to viewer...</div>';

            

            try {

                currentRoom = new LiveKitClient.Room();

                

                currentRoom.on('trackSubscribed', (track, publication, participant) => {

                    if (track.kind === 'video') {

                        mediaContainer.innerHTML = ''; 

                        const videoEl = track.attach();

                        videoEl.className = "w-full h-full object-cover";

                        mediaContainer.appendChild(videoEl);

                    }

                    if (track.kind === 'audio') {

                        track.attach(); 

                    }

                });


                currentRoom.on('disconnected', () => {

                    hideWidget();

                });


                await currentRoom.connect('wss://whutcha-drgxd8d7.livekit.cloud', token);


            } catch (error) {

                console.error("LiveKit connection error:", error);

                mediaContainer.innerHTML = '<div class="text-red-500">Connection Failed</div>';

                setTimeout(hideWidget, 3000);

            }

        }


        async function startListening() {

            const urlParams = new URLSearchParams(window.location.search);

            const streamerId = urlParams.get('streamer') || 'default';

            

            console.log(`OBS Overlay started for streamer: ${streamerId}`);


            // 🔥 1. Загружаем начальную позицию из базы

            const { data: profile } = await supabase

                .from('profiles')

                .select('overlay_position, overlay_scale')

                .eq('id', streamerId)

                .single();

                

            if (profile) {

                applyLayout(profile.overlay_position || 'bottom-right', profile.overlay_scale || 1.0);

            }


            // 🔥 2. Слушаем пульт с айфона (Настройки)

            supabase.channel('obs-layout')

                .on('postgres_changes', { 

                    event: 'UPDATE', 

                    schema: 'public', 

                    table: 'profiles',

                    filter: `id=eq.${streamerId}`

                }, payload => {

                    applyLayout(payload.new.overlay_position, payload.new.overlay_scale);

                })

                .subscribe();

            

            // 3. Слушаем донаты

            supabase

                .channel('obs-overlay-donations')

                .on('postgres_changes', {

                    event: 'UPDATE',

                    schema: 'public',

                    table: 'donations',

                    filter: `status=eq.approved&streamer_id=eq.${streamerId}` 

                }, payload => {

                    const donation = payload.new;

                    console.log("New approved donation:", donation);


                    showWidget(donation);


                    if (donation.video_hash === "costream_request") {

                        if (donation.join_token) {

                            connectCoStream(donation.join_token);

                            setTimeout(() => {

                                if (currentRoom) hideWidget();

                            }, 60000); // Закрываем ко-стрим через минуту

                        }

                    } else {

                        playVideoMessage(donation.video_hash);

                    }

                })

                .subscribe();

        }


        startListening();
